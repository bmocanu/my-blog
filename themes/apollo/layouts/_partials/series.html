{{- /* Navigation for a series: a section whose _index.md sets `series = true`. The posts of the series are ordered by `weight` in their front matter. @context {dict} . `page` is the current page, `mode` is "list" (the box with every part of the series) or "pager" (previous / next links, shown after the post). For "list", `variant` is "side" (sidebar, wide screens) or "inline" (under the post title, narrow screens). @example: {{ partial "series.html" (dict "page" . "mode" "list" "variant" "side") }} */}} {{- $page := .page }} {{- with $page.CurrentSection }} {{- if and .Params.series (ne $page.Kind "section") }} {{- $section := . }} {{- $parts := .RegularPages.ByWeight }} {{- $pos := 0 }} {{- range $i, $p := $parts }} {{- if eq $p $page }}{{ $pos = add $i 1 }}{{ end }} {{- end }} {{- if $pos }} {{- if eq $.mode "list" }}
{{ $section.Title }} series Part {{ $pos }} of {{ len $parts }}
    {{- range $parts }} {{- if eq . $page }}
  1. {{ .Title }}
    2. {{- else }}
  2. {{ .Title }}
    3. {{- end }} {{- end }}
{{- else if eq $.mode "pager" }} {{- $prev := cond (gt $pos 1) (index $parts (sub $pos 2)) nil }} {{- $next := cond (lt $pos (len $parts)) (index $parts $pos) nil }}
{{- with $prev }} ← Previous{{ .Title }} {{- else }} {{- end }} {{- with $next }} Next →{{ .Title }} {{- end }}
{{- end }} {{- end }} {{- end }} {{- end }}
