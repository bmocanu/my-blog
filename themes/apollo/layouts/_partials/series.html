The posts of the series are ordered by `weight` in their front matter. @context {dict} . `page` is the current page, `mode` is "list" (the box with every part of the series) or "pager" (previous / next links, shown after the post). For "list", `variant` is "side" (sidebar, wide screens) or "inline" (under the post title, narrow screens). @example: {{ partial "series.html" (dict "page" . "mode" "list" "variant" "side") }} */}} {{- $page := .page }} {{- with $page.CurrentSection }} {{- if and .Params.series (ne $page.Kind "section") }} {{- $section := . }} {{- /* Upcoming parts (`build: {render: never, list: local}`) are listed but have no page: no link, and the pager skips them. */}} {{- $parts := .RegularPages.ByWeight }} {{- $pos := 0 }} {{- range $i, $p := $parts }} {{- if eq $p $page }}{{ $pos = add $i 1 }}{{ end }} {{- end }} {{- if $pos }} {{- if eq $.mode "list" }}
{{ $section.Title }} series Part {{ $pos }} of {{ len $parts }}
    {{- range $parts }} {{- if eq . $page }}
  1. {{ .Title }}
    2. {{- else if not .RelPermalink }}
  2. {{ .Title }}
    3. {{- else }}
  3. {{ .Title }}
    4. {{- end }} {{- end }}
{{- else if eq $.mode "pager" }} {{- $prev := (where (first (sub $pos 1) $parts) "RelPermalink" "ne" "" | last 1) }} {{- $next := (where (after $pos $parts) "RelPermalink" "ne" "" | first 1) }}

{{- end }} {{- end }} {{- end }} {{- end }}
